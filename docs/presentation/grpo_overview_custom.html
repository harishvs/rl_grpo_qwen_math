    subgraph GENERATE ["1 · Generation (vLLM Server)"]
        direction TB
        P["Prompt
'Half the value of 3x - 9

        VLLM["vLLM Server
(remote, HTTP)"]
        P --> VLLM
        VLLM --> R1["Rollout 1: 'boxed{83}'
✅ reward = 1.0"]
        VLLM --> R2["Rollout 2: '...the correct answer

        VLLM --> R3["Rollout 3: '...the final answer

        VLLM --> R4["Rollout 4: 'We get
boxed{58}'  ❌ reward = 0.0"]
    end

    subgraph REWARD ["2 · Reward (Environment Service)"]
        ENV["Environment Service
(HTTP, binary check)"]
        R1 & R2 & R3 & R4 --> ENV
        ENV --> REW["Rewards: [1.0, 1.0, 0.0, 0.0]"]
    end

    subgraph ADV ["3 · Advantage (GRPO)"]
        GRPO["compute_grpo_advantages
A_i = (r_i − mean) / std"]
        REW --> GRPO
        GRPO --> A1["+0.87"]
        GRPO --> A2["+0.87"]
        GRPO --> A3["−0.87"]
        GRPO --> A4["−0.87"]
    end

    subgraph LOGPROBS ["4 · Log Probs"]
        direction TB
        OLD["old_log_probs
(from vLLM at generation time)
per-token tensors"]
        NEW["new_log_probs
(actor forward pass)
per-token tensors"]
        VLLM -.-> OLD
        ACTOR["Actor Model
(FSDP, current weights)"]
        R1 & R2 & R3 & R4 --> ACTOR --> NEW
    end

    subgraph LOSS ["5 · Policy Update (_update_policy)"]
        direction TB
        RATIO["Per-token importance ratio
ratio = exp(new_lp − old_lp)"]
        OLD & NEW --> RATIO

        CLIP["PPO Clipping (per-token)
min(ratio × A, clip(ratio) × A)
clip range: [0.8, 1.2]"]
        RATIO --> CLIP
        A1 & A2 & A3 & A4 --> CLIP

        KL["KL penalty (per-token, Schulman)
(ratio − 1) − log(ratio)
using old_log_probs as reference"]
        RATIO --> KL

        TOTAL["Total Loss
policy_loss + kl_coef × KL
averaged over tokens, then sequences"]
        CLIP --> TOTAL
        KL --> TOTAL
    end

    subgraph UPDATE ["6 · Weight Update"]
        GRAD["Gradient step
grad_clip → optimizer.step()"]
        TOTAL --> GRAD
        GRAD -->|"every 3 steps"| SYNC["Sync weights → vLLM
(HTTP POST)"]
        SYNC -.-> VLLM
    end

Key difference from textbook GRPO: No separate frozen reference model. KL divergence uses old_log_probs from generation time (the policy that collected the data) instead of a frozen copy of the pretrained model. This saves GPU memory but provides no fixed anchor — the "reference" shifts every step.
